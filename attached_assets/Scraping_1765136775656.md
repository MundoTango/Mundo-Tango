import axios, { AxiosInstance } from 'axios';
import rateLimit from 'axios-rate-limit';
import * as crypto from 'crypto';

export interface ScrapedEvent {
  title: string;
  sourceUrl: string;
  startDate?: Date;
  endDate?: Date;
  city?: string;
  country?: string;
  venue?: string;
  organizerName?: string;
  organizerEmail?: string;
  teachers?: string[];
  djs?: string[];
  price?: string;
  registrationUrl?: string;
  rawHtml: string;
}

export abstract class BaseEventScraper {
  protected name: string;
  protected platform: string;
  protected http: AxiosInstance;
  protected visited: Set<string>;

  constructor(name: string, platform: string, maxRequestsPerMin = 10) {
    this.name = name;
    this.platform = platform;
    this.visited = new Set<string>();
    this.http = rateLimit(axios.create(), {
      maxRequests: maxRequestsPerMin,
      perMilliseconds: 60_000,
    });
  }

  protected abstract scrapeEventList(params: any): Promise<ScrapedEvent[]>;
  protected abstract scrapeEventDetails(url: string): Promise<ScrapedEvent>;
  protected abstract extractLinkedEvents(html: string): Promise<string[]>;

  async scrapeWithRecursion(
    url: string,
    depth = 0,
    maxDepth = 3
  ): Promise<ScrapedEvent[]> {
    if (depth > maxDepth) return [];
    if (this.visited.has(url)) return [];

    this.visited.add(url);
    console.log(`${this.name}: scraping depth=${depth} url=${url}`);

    const event = await this.scrapeEventDetails(url);
    const linkedUrls = await this.extractLinkedEvents(event.rawHtml);

    const childEventsArrays = await Promise.all(
      linkedUrls
        .filter((u) => !this.visited.has(u))
        .map((u) => this.scrapeWithRecursion(u, depth + 1, maxDepth))
    );

    return [event, ...childEventsArrays.flat()];
  }

  protected generateFingerprint(event: ScrapedEvent): string {
    const keyParts = [
      event.title?.trim().toLowerCase() || '',
      event.city?.trim().toLowerCase() || '',
      event.startDate ? event.startDate.toISOString().split('T')[^4_0] : '',
    ];
    const key = keyParts.join('|');
    return crypto.createHash('md5').update(key).digest('hex');
  }

  // TODO: wire to your DB layer
  async saveToDatabase(events: ScrapedEvent[]): Promise<void> {
    for (const event of events) {
      const fingerprint = this.generateFingerprint(event);
