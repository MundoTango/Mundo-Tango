B. Luma Job Submission and Polling

The following pseudo‑code demonstrates how to submit a job to Luma and poll for its completion. Replace fetchLumaApi with your actual HTTP client.

// services/luma.ts
interface LumaJobRequest {
  audioUrl: string;
  phonemes: any[];
  inputMedia?: string;
}

export async function submitLumaJob(req: LumaJobRequest): Promise<string> {
  const response = await fetchLumaApi('/jobs', {
    method: 'POST',
    body: JSON.stringify(req),
  });
  const data = await response.json();
  return data.jobId as string;
}

export async function pollLumaJob(jobId: string): Promise<{ videoUrl: string }> {
  while (true) {
    const res = await fetchLumaApi(`/jobs/${jobId}`);
    const job = await res.json();
    if (job.status === 'completed') {
      return { videoUrl: job.outputUrl };
    }
    if (job.status === 'failed') throw new Error('Luma job failed');
    await new Promise((resolve) => setTimeout(resolve, 5000));
  }
}

1. Map Pin Final Fix
Problem

Event markers on the community map sometimes default to Buenos Aires
(latitude ‑34.6037, longitude ‑58.3816) instead of the correct event
location. Investigation shows that server/routes/event-routes.ts returns
latitude and longitude as strings. When the client code parses
these strings it can produce NaN, triggering a fallback coordinate. The
map-routes.ts endpoint already converts coordinates to numbers, but
event‑detail fetches still return strings.

Solution

Cast latitude/longitude to floats in the event API. Modify
server/routes/event-routes.ts so that any returned event includes
numeric latitude and longitude properties. This can be achieved by
using PostgreSQL’s ::float cast when selecting the fields, or by
parseFloat in JavaScript.

Ensure that null coordinates remain null. Do not default to
zeros, as that will plot the marker in the Gulf of Guinea.

Remove any fallback logic that hardcodes Buenos Aires when
coordinates are missing.

Below is a patch you should apply to the events route. It converts
latitude and longitude to numbers when reading from the database:

// server/routes/event-routes.ts (excerpt)
router.get("/events/:id", async (req, res) => {
  try {
    const id = parseInt(req.params.id);
    const event = await db
      .select({
        id: events.id,
        title: events.title,
        description: events.description,
        // Cast latitude/longitude to float so the client receives numbers
        latitude: db.raw("CAST(latitude AS DOUBLE PRECISION)"),
        longitude: db.raw("CAST(longitude AS DOUBLE PRECISION)"),
        // ... other fields ...
      })
      .from(events)
      .where(eq(events.id, id))
      .limit(1);

    if (!event) {
      return res.status(404).json({ error: "Event not found" });
    }

    res.json(event);
  } catch (error: any) {
    res.status(500).json({ error: error.message });
  }
});

interface TTSWithPhonemes {

async textToSpeechWithTimestamps(
  text: string,
  userId: number,
  voiceId?: string

  const actualVoiceId = await this.getUserVoiceId(userId, voiceId);

      return_phoneme_timestamps: true
    },
    {
      headers: {
        'xi-api-key': this.apiKey,
        'Content-Type': 'application/json'
      },
      responseType: 'json'

  await this.trackUsage(userId, 'tts_phoneme', (charCount / 1000) * 0.30);
  return { audioUrl, phonemeTimestamps: phonemes, characterCount: charCount };
}

   * 1. Converts the text to speech using ElevenLabs (with phonemes).
   * 2. Submits the resulting audio to Luma’s Dream Machine for
   *    audio‑driven generation.
   * 3. Returns the generated video metadata.
   */
  async generateAvatarFromText(userId: number, text: string, aspectRatio: '1:1' | '16:9' = '1:1'): Promise<any> {
    // Step 1: TTS with phonemes
    const ttsResult = await elevenlabsVoiceService.textToSpeechWithTimestamps(text, userId);
    // Step 2: Submit to Luma
    const video = await this.luma.audioToAvatar({
      userId,
      audioUrl: ttsResult.audioUrl,
      prompt: text,
      aspectRatio
    });
    return { video, phonemes: ttsResult.phonemeTimestamps };
  }
}

export const mediaPipelineService = new MediaPipelineService(new LumaService());


Expose a route in server/routes/ai-routes.ts:

// server/routes/ai-routes.ts
import { mediaPipelineService } from '../services/ai/mediaPipeline';

router.post('/ai/avatar-from-text', authenticateToken, async (req: AuthRequest, res) => {
  try {
    const userId = req.userId!;
    const { text, aspectRatio } = req.body;
    const result = await mediaPipelineService.generateAvatarFromText(userId, text, aspectRatio);
    res.json(result);
  } catch (error: any) {
    res.status(500).json({ error: error.message });
  }
});
