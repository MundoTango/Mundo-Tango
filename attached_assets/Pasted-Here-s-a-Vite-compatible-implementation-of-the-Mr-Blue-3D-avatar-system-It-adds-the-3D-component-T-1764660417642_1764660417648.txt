Here’s a Vite‑compatible implementation of the Mr Blue 3D avatar system. It adds the 3D component, TTS integration, an emotion‑detection hook, a new page and a README for model assets. Since the project is built with Vite + Express (not Next.js), environment variables use the VITE_… prefix.

What’s included
File	Purpose
client/src/components/mr-blue/Avatar3D.tsx	Renders the GLB model, handles click‑to‑animate, speech playback, nodding/rotation and integrates with ElevenLabs via a speak() helper.
client/src/lib/ttsService.ts	Wraps the ElevenLabs TTS API; reads VITE_ELEVENLABS_API_KEY and VITE_ELEVENLABS_VOICE_ID from .env, returns audio blobs and provides a useAudio() hook.
client/src/lib/personalityEngine.ts	Stub for future GPT‑4o‑based emotion detection, currently always returns 'idle'.
client/src/hooks/useEmotionDetection.ts	Polls the personality engine at a configurable interval and updates the avatar’s emotion via callback.
client/src/pages/mr-blue-avatar-3d.tsx	A standalone page that hosts the 3D avatar with controls for switching between real/Pixar models and manually setting emotions; also uses the emotion‑detection hook.
public/models/README.md	Explains where to place the GLB model files and their required naming (mr_blue_real.glb, mr_blue_pixar.glb).

// client/src/components/mr-blue/Avatar3D.tsx
import { useRef, useState, useEffect, Suspense } from 'react';
import { Canvas, useFrame } from '@react-three/fiber';
import { OrbitControls, PerspectiveCamera, useGLTF } from '@react-three/drei';
import type { Mesh } from 'three';
import { apiRequest } from '@/lib/queryClient';
import { speak, useAudio } from '@/lib/ttsService';

export interface Avatar3DProps {
  emotion?: 'idle' | 'happy' | 'surprised' | 'nodding';
  autoRotate?: boolean;
  modelType?: 'real' | 'pixar';
  onActivate?: () => void;
}

function AvatarModel({ emotion, autoRotate }: { emotion: Avatar3DProps['emotion']; autoRotate: boolean; }) {
  const meshRef = useRef<Mesh>(null);
  useFrame((state, delta) => {
    if (meshRef.current && autoRotate) {
      meshRef.current.rotation.y += delta * 0.5;
    }
    if (meshRef.current && emotion === 'nodding') {
      meshRef.current.rotation.x = Math.sin(state.clock.elapsedTime * 3) * 0.2;
    }
  });
  return <></>;
}

export default function Avatar3D({
  emotion = 'idle',
  autoRotate = true,
  modelType = 'real',
  onActivate
}: Avatar3DProps) {
  const [currentEmotion, setCurrentEmotion] = useState<Avatar3DProps['emotion']>(emotion);
  const [isLoading, setIsLoading] = useState(false);
  const modelPath = modelType === 'pixar' ? '/models/mr_blue_pixar.glb' : '/models/mr_blue_real.glb';
  const { scene: avatarScene } = useGLTF(modelPath, true) as unknown as { scene: THREE.Group };

  useEffect(() => {
    setCurrentEmotion(emotion);
  }, [emotion]);

  const handleClick = async () => {
    onActivate?.();
    try {
      setIsLoading(true);
      const response = await apiRequest('POST', '/api/mr-blue/agents/avatar/animate', { emotion: 'happy' });
      const data = await response.json();
      if (data.success) {
        setCurrentEmotion('happy');
        setTimeout(() => setCurrentEmotion('idle'), 3000);
      }
    } catch (err) {
      console.error('[Avatar3D] animate error', err);
    } finally {
      setIsLoading(false);
    }
  };

  const audioPlayer = useAudio();
  const handleSpeak = async (text: string) => {
    try {
      setCurrentEmotion('nodding');
      const voiceId = import.meta.env.VITE_ELEVENLABS_VOICE_ID || '';
      const audioBlob = await speak(text, voiceId);
      await audioPlayer(audioBlob);
    } catch (e) {
      console.error('Error speaking:', e);
    } finally {
      setCurrentEmotion('idle');
    }
  };

  return (
    <div className="relative w-full h-full" style={{ minHeight: '200px' }} onClick={handleClick}>
      <Canvas shadows dpr={[1, 2]} gl={{ antialias: true, alpha: true }} className="cursor-pointer">
        <PerspectiveCamera makeDefault position={[0, 0, 5]} />
        <ambientLight intensity={0.5} />
        <pointLight position={[10, 10, 10]} intensity={1} />
        <pointLight position={[-10, -10, -10]} intensity={0.5} color="#4A90E2" />
        <spotLight position={[0, 5, 0]} angle={0.3} penumbra={1} intensity={0.5} castShadow />
        <Suspense fallback={null}>
          {avatarScene && <primitive object={avatarScene} position={[0, -1.2, 0]} />}
          <AvatarModel emotion={currentEmotion} autoRotate={autoRotate && !isLoading} />
        </Suspense>
        <OrbitControls enableZoom={false} enablePan={false} maxPolarAngle={Math.PI / 2} minPolarAngle={Math.PI / 2} />
      </Canvas>
      {isLoading && (
        <div className="absolute inset-0 flex items-center justify-center bg-background/50 backdrop-blur-sm rounded-lg">
          <div className="text-sm text-muted-foreground">Animating...</div>
        </div>
      )}
    </div>
  );
}

// client/src/lib/ttsService.ts
const ELEVENLABS_API_BASE = 'https://api.elevenlabs.io/v1';

export async function speak(text: string, voiceId: string): Promise<Blob> {
  const apiKey = import.meta.env.VITE_ELEVENLABS_API_KEY;
  if (!apiKey) throw new Error('ElevenLabs API key is not configured.');
  if (!voiceId) throw new Error('ElevenLabs voice ID is not configured.');
  const url = `${ELEVENLABS_API_BASE}/text-to-speech/${voiceId}/stream`;
  const response = await fetch(url, {
    method: 'POST',
    headers: {
      'Content-Type': 'application/json',
      'xi-api-key': apiKey
    },
    body: JSON.stringify({
      text,
      model_id: 'eleven_monolingual_v1',
      voice_settings: { stability: 0.5, similarity_boost: 0.75 }
    })
  });
  if (!response.ok) throw new Error(`ElevenLabs API error: ${response.status}`);
  return await response.blob();
}

export function useAudio() {
  return async (audioBlob: Blob) => {
    const url = URL.createObjectURL(audioBlob);
    const audio = new Audio(url);
    await audio.play();
    audio.addEventListener('ended', () => {
      URL.revokeObjectURL(url);
    });
  };
}

// client/src/hooks/useEmotionDetection.ts
import { useEffect } from 'react';
import { analyseUserEmotion, Emotion } from '@/lib/personalityEngine';

export function useEmotionDetection(onEmotionChange: (emotion: Emotion) => void, intervalMs = 5000) {
  useEffect(() => {
    let isMounted = true;
    const poll = async () => {
      try {
        const detected = await analyseUserEmotion();
        if (isMounted) {
          onEmotionChange(detected);
        }
      } finally {
        setTimeout(poll, intervalMs);
      }
    };
    poll();
    return () => {
      isMounted = false;
    };
  }, [onEmotionChange, intervalMs]);
}

            type="radio"
            name="modelType"

        </label>
        <label className="flex items-center gap-2">

          Pixar

<!-- public/models/README.md -->
This directory is intended to store the GLB models used by the Mr Blue 3D avatar.

The project does **not** include any 3D assets by default. To use the 3D avatar component you must supply two `.glb` files with the following names:

- **mr_blue_real.glb** – a realistic human‑style Mr Blue avatar. Ideally this model should have a polycount under 20k, be fully rigged (including facial bones) and include PBR textures.
- **mr_blue_pixar.glb** – a Pixar‑style stylised avatar of Mr Blue. This version should also be rigged and include appropriate materials and textures.

Place the GLB files in this folder so they can be loaded at runtime via `/models/mr_blue_real.glb` or `/models/mr_blue_pixar.glb`. You can source models from third‑party services (e.g. Ready Player Me, Luma AI) or create your own with a 3D modelling tool. Ensure you have the proper license to use and distribute these assets.
