2. REPLACE autoCreateCities() function (line 170-173) with this complete implementation:
```typescript
private async autoCreateCities(): Promise<void> {
  console.log('[Agent #115] Processing scraped events for city matching...');
  
  // Get all scraped events that don't have a group assignment yet
  const unmatchedEvents = await db.query.scrapedEvents.findMany({
    where: sql`${scrapedEvents.groupId} IS NULL`,
    columns: {
      id: true,
      location: true,
      city: true,
    }
  });
  
  console.log(`[Agent #115] Found ${unmatchedEvents.length} events without city groups`);
  
  let matched = 0;
  let newCitiesNeeded: Set<string> = new Set();
  
  for (const event of unmatchedEvents) {
    const locationString = event.location || event.city;
    if (!locationString) continue;
    
    // Try to match to existing city group
    const groupId = await cityMatcherService.matchEventLocation(locationString);
    
    if (groupId) {
      // Update event with matched group
      await db.update(scrapedEvents)
        .set({ groupId })
        .where(eq(scrapedEvents.id, event.id));
      matched++;
    } else {
      // Parse and track genuinely new cities
      const parsed = cityMatcherService.parseLocation(locationString);
      if (parsed.city) {
        newCitiesNeeded.add(parsed.city);
      }
    }
  }
  
  console.log(`[Agent #115] ✅ Matched ${matched} events to existing cities`);
  
  if (newCitiesNeeded.size > 0) {
    console.log(`[Agent #115] 🆕 Detected ${newCitiesNeeded.size} genuinely new cities:`);
    newCitiesNeeded.forEach(city => console.log(`  - ${city}`));
    console.log('[Agent #115] ⚠️  New cities require manual admin approval before creating groups');
  }
}
```
