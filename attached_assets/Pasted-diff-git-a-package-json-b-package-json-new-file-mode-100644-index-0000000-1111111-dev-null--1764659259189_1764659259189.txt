+const ELEVENLABS_API_BASE = 'https://api.elevenlabs.io/v1';
+
+export async function speak(text: string, voiceId: string): Promise<Blob> {
+  const apiKey = process.env.NEXT_PUBLIC_ELEVENLABS_API_KEY;
+  if (!apiKey) {
+    throw new Error('ElevenLabs API key is not configured');
+  }
+  const url = `${ELEVENLABS_API_BASE}/text-to-speech/${voiceId}/stream`;
+  const response = await fetch(url, {
+    method: 'POST',
+    headers: {
+      'Content-Type': 'application/json',
+      'xi-api-key': apiKey
+    },
+    body: JSON.stringify({
+      text,
+      model_id: 'eleven_monolingual_v1',
+      voice_settings: { stability: 0.5, similarity_boost: 0.75 }
+    })
+  });
+  if (!response.ok) {
+    throw new Error(`ElevenLabs API error: ${response.status}`);
+  }
+  const blob = await response.blob();
+  return blob;
+}
+
+export function useAudio() {
+  return async (audioBlob: Blob) => {
+    const url = URL.createObjectURL(audioBlob);
+    const audio = new Audio(url);
+    await audio.play();
+  };
+}

+import { useEffect } from 'react';
+import { analyseUserEmotion, Emotion } from '@/lib/personalityEngine';
+
+export function useEmotionDetection(
+  onEmotionChange: (emotion: Emotion) => void,
+  intervalMs = 5000
+) {
+  useEffect(() => {
+    let isMounted = true;
+    const poll = async () => {
+      const detected = await analyseUserEmotion();
+      if (isMounted) {
+        onEmotionChange(detected);
+      }
+      setTimeout(poll, intervalMs);
+    };
+    poll();
+    return () => {
+      isMounted = false;
+    };
+  }, [onEmotionChange, intervalMs]);
+}

+import { useRef, useState, useEffect, Suspense } from 'react';
+import { Canvas, useFrame } from '@react-three/fiber';
+import { OrbitControls, PerspectiveCamera, useGLTF } from '@react-three/drei';
+import type { Mesh } from 'three';
+import { apiRequest } from '@/lib/queryClient';
+import { speak, useAudio } from '@/lib/ttsService';
+
+export interface Avatar3DProps {
+  emotion?: 'idle' | 'happy' | 'surprised' | 'nodding';
+  autoRotate?: boolean;
+  modelType?: 'real' | 'pixar';
+  onActivate?: () => void;
+}
+
+function AvatarModel({
+  emotion,
+  autoRotate
+}: {
+  emotion: Avatar3DProps['emotion'];
+  autoRotate: boolean;
+}) {
+  const meshRef = useRef<Mesh>(null);
+  useFrame((state, delta) => {
+    if (meshRef.current && autoRotate) {
+      meshRef.current.rotation.y += delta * 0.5;
+    }
+    if (meshRef.current && emotion === 'nodding') {
+      meshRef.current.rotation.x = Math.sin(state.clock.elapsedTime * 3) * 0.2;
+    }
+  });
+  return <primitive ref={meshRef} object={null as any} />;
+}
+
+export default function Avatar3D({
+  emotion = 'idle',
+  autoRotate = true,
+  modelType = 'real',
+  onActivate
+}: Avatar3DProps) {
+  const [currentEmotion, setCurrentEmotion] =
+    useState<Avatar3DProps['emotion']>(emotion);
+  const [isLoading, setIsLoading] = useState(false);
+  const modelPath =
+    modelType === 'pixar'
+      ? '/models/mr_blue_pixar.glb'
+      : '/models/mr_blue_real.glb';
+  const { scene: avatarScene } = useGLTF(modelPath, true) as unknown as {
+    scene: THREE.Group;
+  };
+  useEffect(() => {
+    setCurrentEmotion(emotion);
+  }, [emotion]);
+
+  const handleClick = async () => {
+    onActivate?.();
+    try {
+      setIsLoading(true);
+      const response = await apiRequest(
+        'POST',
+        '/api/mr-blue/agents/avatar/animate',
+        { emotion: 'happy' }
+      );
+      const data = await response.json();
+      if (data.success) {
+        setCurrentEmotion('happy');
+        setTimeout(() => setCurrentEmotion('idle'), 3000);
+      }
+    } catch (err) {
+      console.error('[Avatar3D] animate error', err);
+    } finally {
+      setIsLoading(false);
+    }
+  };
+
+  const audioPlayer = useAudio();
+  const handleSpeak = async (text: string) => {
+    try {
+      setCurrentEmotion('nodding');
+      const audioBlob = await speak(
+        text,
+        process.env.NEXT_PUBLIC_ELEVENLABS_VOICE_ID || ''
+      );
+      await audioPlayer(audioBlob);
+    } catch (e) {
+      console.error('Error speaking:', e);
+    } finally {
+      setCurrentEmotion('idle');
+    }
+  };
+
+  return (
+    <div
+      className="relative w-full h-full"
+      data-testid="avatar-3d-container"
+      style={{ minHeight: '200px' }}
+    >
+      <Canvas
+        shadows
+        dpr={[1, 2]}
+        gl={{ antialias: true, alpha: true }}
+        className="cursor-pointer"
+      >
+        <PerspectiveCamera makeDefault position={[0, 0, 5]} />
+        <ambientLight intensity={0.5} />
+        <pointLight position={[10, 10, 10]} intensity={1} />
+        <pointLight
+          position={[-10, -10, -10]}
+          intensity={0.5}
+          color="#4A90E2"
+        />
+        <spotLight
+          position={[0, 5, 0]}
+          angle={0.3}
+          penumbra={1}
+          intensity={0.5}
+          castShadow
+        />
+        <Suspense fallback={null}>
+          {avatarScene && <primitive object={avatarScene} position={[0, -1.2, 0]} />}
+          <AvatarModel
+            emotion={currentEmotion}
+            autoRotate={autoRotate && !isLoading}
+          />
+        </Suspense>
+        <OrbitControls
+          enableZoom={false}
+          enablePan={false}
+          maxPolarAngle={Math.PI / 2}
+          minPolarAngle={Math.PI / 2}
+        />
+      </Canvas>
+      {isLoading && (
+        <div className="absolute inset-0 flex items-center justify-center bg-background/50 backdrop-blur-sm rounded-lg">
+          <div className="text-sm text-muted-foreground">Animating...</div>
+        </div>
+      )}
+      <div className="absolute bottom-2 left-2 right-2 text-center">
+        <p className="text-xs text-muted-foreground bg-background/80 backdrop-blur-sm rounded px-2 py-1">
+          Click to activate Mr. Blue
+        </p>
+      </div>
+    </div>
+  );
+}
