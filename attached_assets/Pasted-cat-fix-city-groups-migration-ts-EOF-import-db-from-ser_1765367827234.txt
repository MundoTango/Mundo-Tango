async function fixCityGroups() {
  console.log('🚀 Starting city groups migration...');
  
  try {
    // Step 1: Identify venue groups currently marked as cities
    console.log('\n📋 Step 1: Identifying misclassified venue groups...');
    const venueGroups = await db.select().from(groups).where(
      and(
        eq(groups.type, 'city'),
        or(
          like(groups.name, '%Studio%'),
          like(groups.name, '%Dance%'),
          like(groups.name, '%School%'),
          like(groups.name, '%Academy%'),
          like(groups.name, '%Hall%'),
          like(groups.name, '%Church%'),
          like(groups.name, '%Centre%'),
          like(groups.name, '%Center%'),
          like(groups.name, '%House%'),
          like(groups.name, '%Club%'),
          like(groups.name, '%Bar%')
        )
      )
    );
    
    console.log(`Found ${venueGroups.length} venue groups misclassified as cities`);
    venueGroups.forEach(g => console.log(`  - ${g.name} (ID: ${g.id})`));
    
    // Step 2: Reclassify venue groups
    if (venueGroups.length > 0) {
      console.log('\n🔧 Step 2: Reclassifying venue groups to type="venue"...');
      for (const venue of venueGroups) {
        await db.update(groups)
          .set({ type: 'venue' })
          .where(eq(groups.id, venue.id));
        console.log(`  ✓ Reclassified: ${venue.name}`);
      }
    }
    
    // Step 3: Create proper city groups
    console.log('\n🏙️ Step 3: Creating proper city groups...');
    for (const city of TANGO_CITIES) {
      // Check if city already exists
      const existing = await db.select().from(groups).where(
        and(
          eq(groups.type, 'city'),
          eq(groups.name, city.name)
        )
      );
      
      if (existing.length === 0) {
        await db.insert(groups).values({
          name: city.name,
          type: 'city',
          city: city.city,
          country: city.country,
          latitude: city.lat,
          longitude: city.lng,
          description: `Official city group for ${city.name} tango community`,
          isPrivate: false,
          createdAt: new Date(),
        });
        console.log(`  ✅ Created: ${city.name}, ${city.country}`);
      } else {
        console.log(`  ⏭️ Skipped (exists): ${city.name}`);
      }
    }
    
    // Step 4: Validate results
    console.log('\n✅ Step 4: Validating results...');
    const cityCount = await db.select({ count: sql<number>`count(*)` })
      .from(groups)
      .where(eq(groups.type, 'city'));
    
    const venueCount = await db.select({ count: sql<number>`count(*)` })
      .from(groups)
      .where(eq(groups.type, 'venue'));
    
    console.log(`\n📊 Final counts:`);
    console.log(`  - City groups: ${cityCount[0]?.count || 0}`);
    console.log(`  - Venue groups: ${venueCount[0]?.count || 0}`);
    
    console.log('\n🎉 Migration complete! Map should now show only city groups.');
    console.log('\n📍 Next steps:');
    console.log('  1. Restart the app to see changes');
    console.log('  2. Check /community-world-map - should show ~40 cities');
    console.log('  3. Implement scrapers to link events to city groups');
    
  } catch (error) {
    console.error('❌ Migration failed:', error);
    throw error;
  }
}
